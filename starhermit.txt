# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.17.1
engine.minor=19
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The hollow is also steered by dragging on the plaza (mouse or touch); the
# action tray has Undo (Practice), Hint, Boost and Camera buttons.
control.up=KeyW+ArrowUp | Steer up
control.down=KeyS+ArrowDown | Steer down
control.left=KeyA+ArrowLeft | Steer left
control.right=KeyD+ArrowRight | Steer right
control.boost=ShiftLeft+ShiftRight+Space | Boost
control.pause=Escape+KeyP | Pause / resume
control.undo=KeyZ | Undo
control.hint=KeyH | Hint
control.camera=KeyC | Reset camera
engine.group.up=Steering
engine.group.boost=Match

# Gamepad defaults for the same actions (the browser build's pad mapping):
# the left stick or D-pad steers, A or the right trigger boosts, Start
# pauses; X undoes, Y hints, the right stick press resets the camera.
engine.gamepad.up=LeftStickUp+DpadUp
engine.gamepad.down=LeftStickDown+DpadDown
engine.gamepad.left=LeftStickLeft+DpadLeft
engine.gamepad.right=LeftStickRight+DpadRight
engine.gamepad.boost=South+RightTrigger
engine.gamepad.pause=Start
engine.gamepad.undo=West
engine.gamepad.hint=North
engine.gamepad.camera=RightStickPress
